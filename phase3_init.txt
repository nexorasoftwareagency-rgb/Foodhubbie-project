// PHASE 3 INITIALIZATION
// ============================================================================

function _initPhase3Features() {
    // Load settings
    _loadPinnedMessage();
    _loadDisappearingSettings();
    _loadWallpaper();
    
    // Add pinned message banner if exists
    if (_pinnedMessage) {
        _renderPinnedMessageBanner();
    }
    
    // Apply disappearing messages settings
    if (_disappearingDuration > 0) {
        _applyDisappearingMessages();
    }
    
    // Load wallpaper
    _loadWallpaper();
}
